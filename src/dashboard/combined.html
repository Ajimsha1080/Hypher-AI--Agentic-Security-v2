Use Test upstream to confirm the gateway can reach your real MCP server. A refused connection means the upstream server is not running or the URL is wrong.

!

  

  
403  Policy denied

  
Your agent is not allowed to call that tool. Go to My Agents  edit policy  add the tool to the allowlist.

  

  

  

  
!

  

  
403  Prompt injection detected

  
An argument contained a blocked pattern. Check the audit log for the flagged string. If it is a false positive, add an allowlist rule for that tool argument.

  

  

  

  
!

  

  
409  Replay detected

  
Same request sent twice within 5 minutes. This is intentional protection. Change the request ID or add a unique nonce to the arguments.

  

  

  

  
!

  

  
429  Rate limited or agent busy

  
Either your plan's rate limit was hit, or the agent has another request in flight. The SDK auto-retries on 429. For raw HTTP, wait and retry after the Retry-After header value.

How IP allowlists work

  
When enabled, only requests originating from your configured CIDR ranges can call your gateway. Leave empty to allow all IPs (default). Changes take effect within 60 seconds. Uses CIDR notation: 203.0.113.0/24 allows a /24 subnet, 198.51.100.5/32 allows a single IP.

function getAgentApiEndpoint(path) {
  const p = path.startsWith('/') ? path : '/' + path;
  if (location.origin && !location.origin.startsWith('file:')) {
    return p;
  }
  return 'http://localhost:8000' + p;
}

async function loadHitlQueueUI() {
  const container = document.getElementById('hitl-queue-container');
  if (!container) return;
  container.innerHTML = '
Fetching pending approvals...
';
  try {
    let resp;
    try {
      resp = await fetch(getAgentApiEndpoint('/api/v1/approvals'), { headers: { 'X-Tenant-ID': 'tenant_default' } });
    } catch {
      resp = await fetch('http://localhost:8000/api/v1/approvals', { headers: { 'X-Tenant-ID': 'tenant_default' } });
    }
    const items = await resp.json();
    if (!items || !items.length) {
      container.innerHTML = '
No pending high-risk tool approval requests
';
      return;
    }
    container.innerHTML = items.map(item => `
      

        

          

            [${item.risk_level}] ${item.tool_name}
            Req ID: ${item.request_id}
          

          PENDING APPROVAL
        

        
Reason: ${item.reason || 'High-risk tool execution requires human clearance'}

        
Arguments: ${JSON.stringify(item.tool_arguments)}

        

          
          
        

      

    `).join('');
  } catch (err) {
    container.innerHTML = `
Failed to load approvals: ${err.message}
`;
  }
}

async function decideHitlUI(approvalId, decision) {
  try {
    const endpoint = decision === 'approve' ? 'approve' : 'reject';
    let resp;
    try {
      resp = await fetch(getAgentApiEndpoint(`/api/v1/approvals/${approvalId}/${endpoint}`), {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'X-Tenant-ID': 'tenant_default' },
        body: JSON.stringify({ approver_id: 'admin_user', comment: `Decision ${decision} from dashboard UI` })
      });
    } catch {
      resp = await fetch(`http://localhost:8000/api/v1/approvals/${approvalId}/${endpoint}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'X-Tenant-ID': 'tenant_default' },
        body: JSON.stringify({ approver_id: 'admin_user', comment: `Decision ${decision} from dashboard UI` })
      });
    }
    const data = await resp.json();
    alert(`Operation ${decision.toUpperCase()}ED! Decision recorded.`);
    loadHitlQueueUI();
  } catch (err) {
    alert('Decision error: ' + err.message);
  }
}
